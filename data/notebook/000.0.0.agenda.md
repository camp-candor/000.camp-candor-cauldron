### 🕒 HOUR 0-3: THE ALGORITHM OF EXISTENCE (`packages/procgen`)

**Goal:** Create the mathematical laws that determine "What is here?" without looking at a database. We move this logic into a shared package so both the Edge and the Node spine can use it.

**1. The Biome Math**
We use **Simplex Noise** layered with **Masks**.

- **Layer 1 (Continent):** Low frequency noise. Determines Land vs. Ocean.
- **Layer 2 (Elevation):** Medium frequency. Mountains vs. Plains.
- **Layer 3 (Moisture):** Determines Desert vs. Forest.

**File:** `packages/procgen/src/terrain.ts`

```typescript
import { createNoise2D } from "simplex-noise";
import alea from "alea"; // Deterministic PRNG

export class TerrainCompute {
  private noiseHeight: any;
  private noiseMoisture: any;

  constructor(seed: string) {
    // The Seed is the DNA of the Universe.
    // If you change the seed, the entire planet reshapes.
    const rngHeight = alea(`${seed}_height`);
    const rngMoisture = alea(`${seed}_moisture`);
    this.noiseHeight = createNoise2D(rngHeight);
    this.noiseMoisture = createNoise2D(rngMoisture);
  }

  // Pure Math. Zero Lookup.
  public getBiomeAt(
    lat: number,
    lng: number,
  ): "OCEAN" | "MOUNTAIN" | "FOREST" | "DESERT" | "PLAINS" {
    const elevation = this.noiseHeight(lat * 0.5, lng * 0.5);
    const moisture = this.noiseMoisture(lat * 0.8, lng * 0.8);

    // The Logic of Geology
    if (elevation < -0.2) return "OCEAN";
    if (elevation > 0.6) return "MOUNTAIN";

    if (moisture > 0.3) return "FOREST";
    if (moisture < -0.3) return "DESERT";

    return "PLAINS";
  }

  public isHabitable(lat: number, lng: number): boolean {
    const biome = this.getBiome(lat, lng);
    return biome !== "OCEAN" && biome !== "MOUNTAIN";
  }
}
```

### 🕒 HOUR 3-6: THE HAND OF GOD (`apps/edge`)

**Goal:** Create the `WorldGenDO`. This is a **Global Singleton** (or Regional Singleton) responsible for finding a safe place to drop a new soul.

**1. The Search Strategy**
The player does not provide coordinates. The Player says "I want to be born."
The `WorldGenDO` must find a valid H3 index.

- **Algorithm:** Spiral Search.
- **Start:** An arbitrary "Center of the Universe" (e.g., `(0,0)`).
- **Loop:** Check rings of H3 cells outward until `isHabitable()` returns true.

**File:** `apps/edge/src/objects/WorldGenDO.ts`

```typescript
import { DurableObject } from "cloudflare:workers";
import { cellToLatLng, gridDisk } from "h3-js";
import { TerrainCompute } from "@repo/procgen";

export class WorldGenDO extends DurableObject {
  private terrain: TerrainCompute;
  private nextSpawnIndex: number = 0;
  // A predefined "Starting Region" H3 Index (Res 4)
  private readonly GENESIS_REGION = "8428308ffffffff";

  constructor(state: DurableObjectState, env: Env) {
    super(state, env);
    // Initialize Physics with the Global Seed
    this.terrain = new TerrainCompute(env.WORLD_SEED);
  }

  async fetch(request: Request) {
    const url = new URL(request.url);

    if (url.pathname === "/request-spawn") {
      const spawnLocation = await this.findSafeSpawn();
      return new Response(JSON.stringify({ zone_id: spawnLocation }));
    }
    return new Response("World Gen Online");
  }

  private async findSafeSpawn(): Promise<string> {
    // We spiral out from the center of the Genesis Region
    // In a real app, we would cache "Known Good" locations to avoid re-calculating.
    // For now, we search.

    // Get Res 8 children of the Res 4 Genesis Region is expensive...
    // Optimization: Just use a fixed center and spiral kRings.
    const center = "8828308281fffff"; // Arbitrary Center Hex (Res 8)
    let radius = 0;

    // Safety Brake: Don't search forever
    while (radius < 50) {
      const candidates = gridDisk(center, radius);

      // We check the "Ring" (the new hexes added)
      for (const hex of candidates) {
        const [lat, lng] = cellToLatLng(hex);

        // THE CHECK: Is this land?
        if (this.terrain.isHabitable(lat, lng)) {
          // Optional: Check if this zone is "Full" (too many players) via Registry
          // For Day 11, just return it.
          return hex;
        }
      }
      radius++;
    }

    throw new Error("No habitable land found in Genesis Sector.");
  }
}
```

This file intercepts every request to the server. If the user tries to access `/play` without a session, they are bounced. The React Code is **never sent** to unauthenticated users.

**File:** `apps/web/src/middleware.ts`

```typescript
import { clerkMiddleware, createRouteMatcher } from "@clerk/astro/server";

// DEFINE PROTECTED ROUTES
const isProtectedRoute = createRouteMatcher(["/play(.*)"]);

export const onRequest = clerkMiddleware((auth, context) => {
  // THE CHECK
  if (isProtectedRoute(context.request)) {
    const { userId } = auth();

    // THE REJECTION
    if (!userId) {
      return auth().redirectToSignIn();
    }
  }

  return context.next();
});
```

```typescript
import { ZoneSnapshot, MutationLog, EntitySpawn } from "@repo/schema";

export class StateEngine {
  // The Source of Truth (RAM)
  private entities: Map<string, EntitySpawn> = new Map();
  private tick: number = 0;

  constructor() {}

  // HYDRATION: Loading from the Tomb
  public loadSnapshot(snapshot: ZoneSnapshot) {
    this.tick = Number(snapshot.lastTick);
    for (const entity of snapshot.entities) {
      this.entities.set(entity.entityId, entity);
    }
  }

  // MUTATION: Applying History
  public applyLog(log: MutationLog) {
    if (log.spawn) {
      this.entities.set(log.spawn.entityId, log.spawn);
    } else if (log.update) {
      const target = this.entities.get(log.update.entityId);
      if (target) {
        // Merge deltas
        target.state = { ...target.state, ...log.update.delta };
      }
    } else if (log.despawn) {
      this.entities.delete(log.despawn.entityId);
    }
    this.tick = Number(log.tick);
  }

  public serialize(): ZoneSnapshot {
    return {
      lastTick: BigInt(this.tick),
      entities: Array.from(this.entities.values()),
      environmentOverrides: {}, // Implement later
    };
  }
}
```

```typescript
  private async hydrate() {
    // 1. Load Snapshot
    const snapshotBin = await this.storage.get("snapshot_latest");
    if (snapshotBin) {
      const snapshot = ZoneSnapshot.decode(
        new Uint8Array(snapshotBin as ArrayBuffer),
      );
      this.kernel.loadSnapshot(snapshot);
```

```typescript
import { z } from "zod";

const LogLevel = z.enum(["DEBUG", "INFO", "WARN", "ERROR", "FATAL"]);

export const LogSchema = z.object({
  level: LogLevel,
  trace_id: z.string().uuid(),
  zone_id: z.string().optional(),
  actor_id: z.string().optional(),
  event: z.string(),
  meta: z.record(z.any()),
  timestamp: z.number(),
});

export type LogEntry = z.infer<typeof LogSchema>;

export class KernelLogger {
  private context: Partial<LogEntry>;

  constructor(context: Partial<LogEntry> = {}) {
    this.context = context;
  }

  log(level: z.infer<typeof LogLevel>, event: string, meta: any = {}) {
    const entry: LogEntry = {
      level,
      trace_id: this.context.trace_id || crypto.randomUUID(),
      timestamp: Date.now(),
      event,
      meta,
      ...this.context,
    };
    // In Edge: Prints JSON string for Cloudflare Tail to pick up.
    // In Node: Prints formatted JSON.
    console.log(JSON.stringify(entry));
  }

  withContext(ctx: Partial<LogEntry>) {
    return new KernelLogger({ ...this.context, ...ctx });
  }
}
```

```typescript
  async fetch(request: Request) {
    // INGEST: Map Phase (From Zones)

        new Uint8Array(await request.arrayBuffer()),
      );
      this.aggregate(report);
      return new Response("Recorded");
    }
    return new Response("Bank Open");

  async alarm() {
    // REDUCE: Calculate new prices
    const update = this.calculateMarketCorrection();

    // BROADCAST: Push to Region
    const regionId = this.env.REGION_DO.idFromName(this.state.id.toString());
    const region = this.env.REGION_DO.get(regionId);

    const payload = MarketUpdate.encode(update).finish();
    await region.fetch("http://internal/broadcast/market", {
      method: "POST",
      body: payload,
    });

    // Loop
    await this.state.storage.setAlarm(Date.now() + 60000);
  }

  private aggregate(report: EconomicReport) {
    // Simplified Logic: Add deltas to pools
    for (const [tag, delta] of Object.entries(report.commodity_deltas)) {
      const current = this.commodities.get(tag) || { pool: 1000, demand: 100 };
      // Negative delta = removed from world (bought) = supply down
      current.pool += delta;
      this.commodities.set(tag, current);
    }
  }

  private calculateMarketCorrection(): MarketUpdate {
    const multipliers: Record<string, number> = {};
    for (const [tag, data] of this.commodities) {
      // Scarcity Logic: If Pool < Demand, Price Goes Up
      const ratio = data.demand / Math.max(1, data.pool);
      multipliers[tag] = Math.max(0.1, Math.min(5.0, ratio));
    }
    return {
      tick: BigInt(Date.now()),
      globalInflation: this.inflation,
      priceMultipliers: multipliers,
      taxRates: {},
    };
  }
```

```tsx
export const Stage = () => {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  // THE RENDER LOOP
  useEffect(() => {
    let frameId: number;

    const render = () => {
      const ctx = canvasRef.current?.getContext("2d");
      const snapshots = useGameStore.getState().snapshots;

      if (snapshots.length >= 2) {
        const prev = snapshots[snapshots.length - 2];
        const next = snapshots[snapshots.length - 1];

        // Calculate Interpolation Factor (Alpha)
        // Alpha = (Now - Prev.Time) / (Next.Time - Prev.Time)
        const now = Date.now(); // Note: Needs server-time offset sync in real prod
        const alpha = Math.min(1, (now - prev.tick) / (next.tick - prev.tick));

        // DRAW
        ctx?.clearRect(0, 0, 800, 600);
        next.entities.forEach((entity, id) => {
          const prevEnt = prev.entities.get(id);
          if (prevEnt) {
            const x = prevEnt.x + (entity.x - prevEnt.x) * alpha;
            const y = prevEnt.y + (entity.y - prevEnt.y) * alpha;
            drawSprite(ctx, entity.type, x, y);
          }
        });
      }

      frameId = requestAnimationFrame(render);
    };

    render();
    return () => cancelAnimationFrame(frameId);
  }, []);
```

_The Geometry of Connection._

We do not use a load balancer. We use **Math**.

**File:** `src/router.ts`

```typescript
import { h3ToParent, geoToH3 } from "h3-js";
import { PlayerIntent } from "./proto/codex"; // Compiled Protobuf

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);

    // ROUTE 1: WEBSOCKET CONNECT (Client -> Nomad)
    if (url.pathname === "/connect") {
      const upgradeHeader = request.headers.get("Upgrade");
      if (!upgradeHeader || upgradeHeader !== "websocket") {
        return new Response("Expected Upgrade: websocket", { status: 426 });
      }

      // 1. Authenticate (Clerk/JWT check omitted for brevity)
      const userId = url.searchParams.get("user_id");

      // 2. Deterministic Nomad ID
      // The Nomad ID is stable. It is the User ID.
      const nomadId = env.NOMAD_DO.idFromName(userId);
      const nomadStub = env.NOMAD_DO.get(nomadId);

      return nomadStub.fetch(request);
    }

    // ROUTE 2: ZONE LOOKUP (Internal Use)
    // "Where am I?"
    if (url.pathname === "/zone/resolve") {
      const lat = parseFloat(url.searchParams.get("lat"));
      const lng = parseFloat(url.searchParams.get("lng"));

      // H3 Resolution 8: The Zone Size (~0.7km^2)
      const zoneH3 = geoToH3(lat, lng, 8);
      return new Response(zoneH3);
    }

    return new Response("Codex Router Online", { status: 200 });
  },
};
```

### 1.3 The Atomic Isolate (`NomadDO` & `ZoneDO`)

_The Bi-Directional Tunnel._

The **Nomad** holds the socket. The **Zone** holds the physics.

**File:** `src/objects/NomadDO.ts`

```typescript
export class NomadDO {
  state: DurableObjectState;
  env: Env;
  currentZoneStub: DurableObjectStub | null = null;

  constructor(state: DurableObjectState, env: Env) {
    this.state = state;
    this.env = env;
  }

  async fetch(request: Request) {
    // Handle WebSocket Upgrade
    if (request.headers.get("Upgrade") === "websocket") {
      const pair = new WebSocketPair();
      const [client, server] = Object.values(pair);

      this.state.acceptWebSocket(server);
      return new Response(null, { status: 101, webSocket: client });
    }
    return new Response("Nomad Active");
  }

  async webSocketMessage(ws: WebSocket, message: ArrayBuffer) {
    // 1. Decode Protobuf (Input)
    // const intent = PlayerIntent.decode(new Uint8Array(message));

    // 2. Locate Zone (If not already connected)
    // For MVP, we assume a static starting zone or fetch from storage
    if (!this.currentZoneStub) {
      // Hardcoded start for Day 1: "8828308281fffff"
      const zoneId = this.env.ZONE_DO.idFromName("8828308281fffff");
      this.currentZoneStub = this.env.ZONE_DO.get(zoneId);
    }

    // 3. Proxy to Zone (RPC via Service Binding)

    ws.send(mutationBlob);
  }
}
```
