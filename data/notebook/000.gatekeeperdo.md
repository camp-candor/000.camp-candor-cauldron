**1. The Auth Logic**
**File:** `packages/guard/src/auth.ts`

```typescript
import * as jose from 'jose';

// THE LAW: TRUST NO ONE
// We use RS256 for external (Clerk) and HS256 for internal (Speed).

const JWKS_URL = new URL('https://your-clerk-domain.clerk.accounts.dev/.well-known/jwks.json');
const INTERNAL_SECRET = new TextEncoder().encode('SUPER_SECRET_INTERNAL_KEY_DO_NOT_LEAK');

export class Guard {
  private JWKS: ReturnType<typeof jose.createRemoteJWKSet>;

  constructor() {
    this.JWKS = jose.createRemoteJWKSet(JWKS_URL);
  }

  // 1. VERIFY EXTERNAL (Slow, performed only at Gatekeeper)
  async verifyExternalUser(jwt: string) {
    try {
      const { payload } = await jose.jwtVerify(jwt, this.JWKS);
      return payload.sub; // The User ID
    } catch (e) {
      return null;
    }
  }

  // 2. MINT INTERNAL PASS (Fast, symmetric)
  // This token proves the user passed the Rate Limit check.
```

```typescript
interface Bucket {
  tokens: number;
  lastRefill: number;
}

export class RateLimiter {
  private buckets: Map<string, Bucket> = new Map();
  private readonly capacity: number;
  private readonly refillRate: number; // tokens per second

  constructor(capacity = 10, refillRate = 1) {
    this.capacity = capacity;
    this.refillRate = refillRate;
  }

  check(id: string): boolean {
    const now = Date.now();
    let bucket = this.buckets.get(id);

    if (!bucket) {
      bucket = { tokens: this.capacity, lastRefill: now };
      this.buckets.set(id, bucket);
    }

    // 1. REFILL (Lazy)
    const elapsed = (now - bucket.lastRefill) / 1000;
    const refill = Math.floor(elapsed * this.refillRate);

    if (refill > 0) {
      bucket.tokens = Math.min(this.capacity, bucket.tokens + refill);
      bucket.lastRefill = now;
    }

    // 2. CONSUME
    if (bucket.tokens >= 1) {
      bucket.tokens -= 1;
      return true; // Allowed
    }

    return false; // REJECTED
  }

  // Garbage Collection for memory hygiene
  prune(ttlSeconds: number) {
    const now = Date.now();
    for (const [id, bucket] of this.buckets) {
      if (now - bucket.lastRefill > ttlSeconds * 1000) {
        this.buckets.delete(id);
      }
    }
  }
}
```

```typescript
// We use Web Crypto API (Native to V8/Workers) for maximum speed.
// HMAC-SHA256 is much faster than RSA verification.

export class TicketMaster {
  private secret: CryptoKey | null = null;

  constructor(private secretStr: string) {}

  private async getKey(): Promise<CryptoKey> {
    if (this.secret) return this.secret;
    const enc = new TextEncoder();
    this.secret = await crypto.subtle.importKey('raw', enc.encode(this.secretStr), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign', 'verify']);
    return this.secret;
  }

  // ISSUE TICKET (Performed by Gatekeeper)
  async issue(userId: string, shardId: string): Promise<string> {
    const key = await this.getKey();
    const exp = Date.now() + 30000; // 30 Seconds to connect
    const payload = `${userId}:${shardId}:${exp}`;

    const signature = await crypto.subtle.sign('HMAC', key, new TextEncoder().encode(payload));

    // Convert binary signature to Base64
    const sigStr = btoa(String.fromCharCode(...new Uint8Array(signature)));

    return `${payload}.${sigStr}`; // FORMAT: Data.Signature
  }

  // VERIFY TICKET (Performed by Nomad/Router)
  async verify(ticket: string): Promise<{ userId: string; shardId: string } | null> {
    try {
      const [payload, sigStr] = ticket.split('.');
      if (!payload || !sigStr) return null;

      const key = await this.getKey();
      const signature = Uint8Array.from(atob(sigStr), (c) => c.charCodeAt(0));

      const isValid = await crypto.subtle.verify('HMAC', key, signature, new TextEncoder().encode(payload));

      if (!isValid) return null;

      const [userId, shardId, expStr] = payload.split(':');
      if (Date.now() > parseInt(expStr)) return null; // Expired

      return { userId, shardId };
    } catch (e) {
      return null;
    }
  }
}
```

### 🕒 HOUR 3-6: THE POET (`NarrativeDO`)

**Goal:** Create the Durable Object that holds the context window, debounces events, and calls the LLM.

**1. The Narrative Object**
**File:** `apps/edge/src/objects/NarrativeDO.ts`

```typescript
import { DurableObject } from 'cloudflare:workers';

interface ContextItem {
  role: 'system' | 'user' | 'assistant';
  content: string;
}

export class NarrativeDO extends DurableObject {
  // THE BUFFER
  // We hold events here to debounce them (See Pillar III)
  eventBuffer: any[] = [];
  isProcessing: boolean = false;

  // THE MEMORY
  // A sliding window of chat history
  history: ContextItem[] = [];

  constructor(ctx: DurableObjectState, env: Env) {
    super(ctx, env);
    // Initialize system prompt
    this.history.push({
      role: 'system',
      content: 'You are a Narrator. Describe the events provided. Do not invent mechanics. Be brief and visceral.',
    });
  }

  async fetch(request: Request) {
    const url = new URL(request.url);

    // ROUTE: INGEST FACT
    // Called by ZoneDO when physics happen
    if (url.pathname === '/ingest') {
      const fact = await request.json();
      this.eventBuffer.push(fact);

      // THE DEBOUNCE (Simple Version)
      // If not already thinking, start thinking in 500ms
      if (!this.isProcessing) {
        this.ctx.waitUntil(this.processQueue());
      }
      return new Response('Ack');
    }

    return new Response('Narrator Online');
  }

  async processQueue() {
    this.isProcessing = true;

    // 1. Wait for the "Action Cluster" to finish
    await new Promise((r) => setTimeout(r, 500));

    // 2. Snapshot the buffer
    if (this.eventBuffer.length === 0) {
      this.isProcessing = false;
      return;
    }

    const batch = [...this.eventBuffer];
    this.eventBuffer = []; // Clear for next batch

    // 3. Construct the Prompt
    const prompt = batch.map((f) => `Action: ${f.actor_name} performed ${f.action} on ${f.target_name}. ` + `Result: ${f.success ? 'SUCCESS' : 'FAILURE'} (${f.reason}). ` + `Value: ${f.value}. Context: ${f.tags.join(', ')}.`).join('\n');

    // 4. Call LLM (Mocked for Day 7 Structure, implement real fetch to OpenAI here)
    await this.callLLM(prompt, batch[0].nomad_id); // Reply to the initiator (simplified)

    this.isProcessing = false;
  }

  async callLLM(prompt: string, nomadId: string) {
    // SIMULATING OPENAI STREAM
    // In prod, use fetch('https://api.openai.com/...') with stream:true
    const mockResponse = `The attempt to strike fails! The armor deflects the blow with a harsh clang.`;

    const nomadStub = this.env.NOMAD_DO.get(this.env.NOMAD_DO.idFromName(nomadId));

    // Stream it character by character (Simulating token stream)
    for (const char of mockResponse.split('')) {
      await nomadStub.fetch('http://internal/stream', {
        method: 'POST',
        body: JSON.stringify({ text: char, is_end: false }),
      });
    }

    // End Stream
    await nomadStub.fetch('http://internal/stream', {
      method: 'POST',
      body: JSON.stringify({ text: '', is_end: true }),
    });
  }
}
```
