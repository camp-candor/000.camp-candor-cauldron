**2. The Shader Enforcer (The Filter of Sin)**
**File:** `apps/renderer/src/shaders/JudgmentPost.ts`

```glsl
uniform float u_Time;
uniform sampler2D u_Scene;
uniform vec3 u_ShadowColor; // #000000
uniform float u_GrainAmount; // 0.35

void main() {
    vec4 color = texture2D(u_Scene, v_Uv);

    // 1. THE CRUSH
    // Any pixel below 0.2 brightness is sent to the Void.
    // Darkness is not a gradient; it is a cliff.
    if (length(color.rgb) < 0.2) {
        color.rgb = u_ShadowColor;
    }

    // 2. THE GRAIN OF DUST
    // The world is old and abraded.
    float noise = fracture(sin(dot(v_Uv + u_Time, vec2(12.9898, 78.233))) * 43758.5453);
    color.rgb += (noise - 0.5) * u_GrainAmount;

    // 3. THE EXPOSURE
    // Highlights are blown out. No details in the light.
    if (length(color.rgb) > 0.9) {
        color.rgb = vec3(1.0, 1.0, 1.0); // Pure Blinding White
    }

    gl_FragColor = color;
}
```
