### 🕒 HOUR 0-3: THE TRIPLE STORE KERNEL (`packages/graph-native`)

**Goal:** Create a lightweight, type-safe Graph database that runs inside the V8 Isolate. We cannot use heavy libraries with binary dependencies (like LevelDB) inside a Worker. We build a **native `Map`-based Triple Store**.

**1. The Workspace**

**2. The In-Memory Graph Engine**
We need to store `(Subject, Predicate, Object)`.
**File:** `packages/graph-native/src/MemoryGraph.ts`

```typescript
export type Triple = {
  subject: string; // "player_grog"
  predicate: string; // "WIELDS"
  object: string; // "item_sword"
  meta?: Record<string, any>; // { hand: "right" }
};

export class MemoryGraph {
  // Indices for O(1) lookups
  // subject -> predicate -> object[]
  private indexS: Map<string, Map<string, Set<string>>> = new Map();
  // object -> predicate -> subject[] (Reverse lookup)
  private indexO: Map<string, Map<string, Set<string>>> = new Map();

  // Metadata storage: "S:P:O" -> Data
  private meta: Map<string, any> = new Map();

  constructor() {}

  // ATOMIC WRITE
  insert(s: string, p: string, o: string, meta?: any) {
    this.addToIndex(this.indexS, s, p, o);
    this.addToIndex(this.indexO, o, p, s);
    if (meta) this.meta.set(`${s}:${p}:${o}`, meta);
  }

  // ATOMIC DELETE
  remove(s: string, p: string, o: string) {
    this.removeFromIndex(this.indexS, s, p, o);
    this.removeFromIndex(this.indexO, o, p, s);
    this.meta.delete(`${s}:${p}:${o}`);
  }

  // QUERY: "What is Grog wielding?"
  query(s: string, p: string): string[] {
    const preds = this.indexS.get(s);
    if (!preds) return [];
    const objs = preds.get(p);
    return objs ? Array.from(objs) : [];
  }

  // QUERY: "Who is wielding the sword?"
  reverseQuery(o: string, p: string): string[] {
    const preds = this.indexO.get(o);
    if (!preds) return [];
    const subjs = preds.get(p);
    return subjs ? Array.from(subjs) : [];
  }

  // QUERY: "Get Metadata"
  getMeta(s: string, p: string, o: string) {
    return this.meta.get(`${s}:${p}:${o}`);
  }

  // Helper for internal Map management
  private addToIndex(index: Map<string, any>, key1: string, key2: string, value: string) {
    if (!index.has(key1)) index.set(key1, new Map());
    const level2 = index.get(key1);
    if (!level2.has(key2)) level2.set(key2, new Set());
    level2.get(key2).add(value);
  }

  // Serialize for Persistence
  export() {
    // Logic to dump triples to JSON for storage
  }
}
```

    // 1. FIND CURRENT LOCATION
    // Query: (*)-[:LOCATED_AT]->(Item)
    // Reverse lookup to find who holds it now (Room? Player?)
    const currentHolders = this.localGraph.reverseQuery(itemId, 'CONTAINS');

    for (const holder of currentHolders) {
      this.localGraph.remove(holder, 'CONTAINS', itemId);
    }

    this.localGraph.insert(newOwnerId, 'CONTAINS', itemId, { timestamp: Date.now() });

**1. The Deontic Engine**

```typescript
import { MemoryGraph } from '@repo/graph-native';

export class DeonticEngine {
  private graph: MemoryGraph;

  constructor(graph: MemoryGraph) {
    this.graph = graph;
  }

  // THE CHECK: "Can I enter Room B?"
  canTraverse(actorId: string, fromRoom: string, toRoom: string): { allowed: boolean; reason?: string } {
    // 1. Find the connection (Edge)
    // (RoomA)-[:CONNECTS_TO]->(RoomB)
    // We check metadata on the edge.
    const meta = this.graph.getMeta(fromRoom, 'CONNECTS_TO', toRoom);

    // 2. ALETHIC CHECK (Does the path exist?)
    if (!meta) {
      return { allowed: false, reason: 'NO_PATH' };
    }

    // 3. DEONTIC CHECK (Is it locked?)
    if (meta.state === 'LOCKED') {
      // 4. KEY CHECK
      // Does the actor possess the key?
      // Query: (Actor)-[:CONTAINS]->(Key)
      const keyId = meta.key_id;
      const inventory = this.graph.query(actorId, 'CONTAINS');

      if (inventory.includes(keyId)) {
        return { allowed: true, reason: 'KEY_USED' };
      }

      return { allowed: false, reason: 'LOCKED' };
    }

    return { allowed: true };
  }
}
```

import { MagicPhysics } from '@repo/types';
import { Entity, MutationResult } from '../types';

export class MagicSystem {
  constructor(private config: MagicPhysics) {}

  // The Gatekeeper
  canCast(caster: Entity, spellCost: number): boolean {
    if (this.config.saturation <= 0) return false; // Dead Magic Zone

    switch (this.config.resource_model) {
      case 'SLOTS':
        return caster.slots >= spellCost;
      case 'HEALTH':
        return caster.hp > spellCost;
      case 'POINTS':
        return caster.mana >= spellCost;
      case 'ENTROPY':
        return true; // You can always cast, but it might kill you
    }
  }

  // The Executioner
  processCast(caster: Entity, spell: any): MutationResult {
    const result = new MutationResult();

    // 1. PAY THE COST
    this.deductResource(caster, spell.cost, result);

    // 2. CHECK VOLATILITY (The Wild Magic Check)
    if (Math.random() < this.config.volatility_index) {
      return this.triggerWildSurge(caster, result);
    }

    // 3. CHECK ENVIRONMENT (The Defiling Check)
    if (this.config.environmental_decay > 0) {
      this.triggerDecay(caster, spell.power, result);
    }

    // 4. RESOLVE SPELL
    // Apply effects...

    return result;
  }

  private triggerDecay(caster: Entity, power: number, result: MutationResult) {
    // Determine blast radius based on config
    const radius = power * this.config.environmental_decay;

    result.addEnvironmentMutation({
      center: caster.h3Index,
      radius: radius,
      type: 'WITHER_VEGETATION',
    });

    // In Dark Sun, defiling restores mana or buffs the spell
    result.addBuff(caster, 'ARCANE_SURGE');
  }
}
