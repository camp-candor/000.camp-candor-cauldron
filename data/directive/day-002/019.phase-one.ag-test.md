const fs = require('fs');

// Extract declared lst items from menu.library.ts
const lstMatch = content.match(/lst\s*=\s*\[([\s\S]*?)\];/);
if (!lstMatch) { console.error('::error:: Unable to extract lst declaration'); process.exit(1); }

const rawLst = lstMatch[1]
  .split('\n')
  .map(line => line.replace(/\/\/.*$/, '').trim())
  .filter(line => line && !line.startsWith('//'))
  .map(line => line.replace(/,$/, '').replace(/Act[A-Za-z]+\./, '').replace(/\.split\('\\]'\)\[1\]/, '').replace(/['\"]/g, '').trim());

// Extract all switch case branches
const cases = [...content.matchAll(/case\s+([^:]+):/g)].map(m => 
  m[1].replace(/Act[A-Za-z]+\./, '').replace(/\.split\('\\]'\)\[1\]/, '').replace(/['\"]/g, '').trim()
);

// Verify every item in lst has a corresponding case
const missingCases = rawLst.filter(item => !cases.includes(item));
if (missingCases.length > 0) {
  console.error('::error:: Dead-end routes in lst lacking case handlers:', missingCases);
  process.exit(1);
}

// Verify no duplicate cases exist across the entire switch block
const counts = {};
cases.forEach(c => { counts[c] = (counts[c] || 0) + 1; });
const duplicates = Object.entries(counts).filter(([_, count]) => count > 1);
if (duplicates.length > 0) {
  console.error('::error:: Duplicate case labels detected:', duplicates);
  process.exit(1);
}

console.log('>> [OK] All ' + rawLst.length + ' menu options map 1:1 to unique, distinct switch cases.');
