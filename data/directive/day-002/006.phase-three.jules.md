import HardwareBit, { RigIdentityType } from '../fce/hardware.bit.js';

import State from '../../99.core/state.js';

export const selectHardwareProfile = async (cpy: HardwareModel, bal: HardwareBit, ste: State): Promise<HardwareModel> => {
  const options = [
    'RIG 1: Conductor (CPU / High RAM / Audio Clockmaster / Orchestrator)',
    'RIG 2: Foundry (GPU / High VRAM / PyTorch / Diffusion Worker)'
  ];

  if (bal.src) {
    const raw = bal.src.toUpperCase();
    cpy.selectedRig = raw.includes('RIG_2') || raw.includes('FOUNDRY') ? 'RIG_2_FOUNDRY' : 'RIG_1_CONDUCTOR';
  } else if (global.LIBRARY && typeof global.LIBRARY.hunt === 'function') {
    const choiceBit = await global.LIBRARY.hunt('[Open action] Open Choice', {
      idx: 'choice-hardware',
      title: 'SELECT WORKSTATION HARDWARE PROFILE',
      lst: options
    });

    const chosen = choiceBit?.chcBit?.src || options[0];
    cpy.selectedRig = chosen.startsWith('RIG 2') ? 'RIG_2_FOUNDRY' : 'RIG_1_CONDUCTOR';
  } else {
    cpy.selectedRig = 'RIG_1_CONDUCTOR';
  }

  cpy.configuredAt = new Date().toISOString();

  if (global.LIBRARY && typeof global.LIBRARY.hunt === 'function') {
    try {
      await global.LIBRARY.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [HARDWARE] Profile selected: ${cpy.selectedRig}`
      });
    } catch {
      // Non-blocking fallback
    }
  }

  if (bal.slv) {
    bal.slv({
      hrdBit: {
        idx: 'select-hardware',
        src: cpy.selectedRig,
        val: cpy.selectedRig === 'RIG_2_FOUNDRY' ? 2 : 1,
        dat: {
          rig_identity: cpy.selectedRig,
          configuredAt: cpy.configuredAt
        }
      }
    });
  }
  return cpy;
};

import * as fs from 'fs';
import * as path from 'path';
import ConfigBit, { CauldronConfigFile } from '../fce/config.bit.js';
import { ConfigModel } from '../config.model.js';
import State from '../../99.core/state.js';

export const stampHardwareConfiguration = async (cpy: ConfigModel, bal: ConfigBit, ste: State): Promise<ConfigModel> => {
  const rigProfile = (bal.src || 'RIG_1_CONDUCTOR').toUpperCase();
  const isFoundry = rigProfile.includes('RIG_2') || rigProfile.includes('FOUNDRY');
  const targetRig = isFoundry ? 'RIG_2_FOUNDRY' : 'RIG_1_CONDUCTOR';

  const rootDir = bal.dat?.customRoot || process.cwd();
  const targetPath = path.join(rootDir, 'cauldron.config.json');
  const tempPath = path.join(rootDir, 'cauldron.config.json.tmp');

  const configPayload: CauldronConfigFile = {
    rig_identity: targetRig,
    audio_port: targetRig === 'RIG_1_CONDUCTOR' ? 8000 : 8001,
    gpu_acceleration: targetRig === 'RIG_2_FOUNDRY',
    max_memory_concurrency: targetRig === 'RIG_2_FOUNDRY' ? 4 : 1,
    updated_at: new Date().toISOString()
  };

  // Atomic POSIX write: stage to .tmp first then atomic rename
  fs.writeFileSync(tempPath, JSON.stringify(configPayload, null, 2), { encoding: 'utf8' });
  fs.renameSync(tempPath, targetPath);

  cpy.lastStampedConfig = configPayload;

  if (global.LIBRARY && typeof global.LIBRARY.hunt === 'function') {
    try {
      await global.LIBRARY.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [CONFIG STAMP] Serialized profile ${targetRig} to cauldron.config.json`
      });
    } catch {
      // Non-blocking fallback
    }
  }

  if (bal.slv) {
    bal.slv({
      cfgBit: {
        idx: 'stamp-config',
        val: 1,
        dat: configPayload
      }
    });
  }
  return cpy;
};
