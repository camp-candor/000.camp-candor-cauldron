### II. ARCHITECTURAL MISSION & REQUIREMENTS

In Blessed curses applications, forms and lists attach global event handlers to the shared terminal `screen` instance[cite: 1, 2]. When operators navigate through nested choice lists (e.g., `LIBRARY MENU -> UPDATE UNIT -> SELECT PIVOT`), failing to detach event listeners produces "zombie" callbacks that intercept keystrokes intended for subsequent panels, causes memory leaks, and leaks visual artifacts onto the terminal buffer[cite: 1, 2].

Your mission is to harden `85.choice.unit` and its test battery:
1. **Enforce Atomic Listener Teardown in `openChoice`:**
   - Bind vertical spatial keys (`['up', 'k']` and `['down', 'j']`) using named listener references (`handleUp`, `handleDown`)[cite: 1, 2].
   - Maintain an idempotent `cleanup()` closure that unbinds key handlers via `screen.unkey`, executes `form.destroy()`, and repaints via `screen.render()` prior to resolving `bal.slv`[cite: 1, 2].
   - Handle both form submission (`form.on('submit')`) and cancellation (`form.on('cancel')`) gracefully[cite: 1, 2].
2. **Remove Unused/Dormant Debugger Statements:**
   - Ensure `initChoice` and `updateChoice` are pure, non-blocking handlers resolving immediately with zero `debugger;` statements[cite: 2, 6].
3. **Expand the AVA Test Gauntlet (`choice.buzz.test.ts`):**
   - Retain existing headless safety and unkey unit tests[cite: 2].
   - Implement a re-entrant listener leak gauntlet that simulates 100 sequential choice mount/submit cycles and proves that remaining listener registrations for `up`, `down`, `k`, and `j` remain strictly at `0` while widget destruction reaches `100/100`[cite: 1, 2].
   - Verify that focus callbacks (`dat.cb`) trigger properly when list items are focused[cite: 1, 9].
   - Verify that empty item lists (`lst: []`) execute cleanly without throwing exceptions[cite: 2].

import type { ChoiceModel } from '../choice.model'
import type ChoiceBit from '../fce/choice.bit'
import type State from '../../99.core/state'
import type NetBit from '../../81.grid.unit/fce/net.bit'
import * as Color from '../../val/console-color'

export const initChoice = (cpy: ChoiceModel, bal: ChoiceBit, _ste: State) => {
    if (bal?.slv != null) {
        bal.slv({ chcBit: { idx: 'init-choice' } })
    }
    return cpy
}

export const updateChoice = (cpy: ChoiceModel, bal: ChoiceBit, _ste: State) => {
    if (bal?.slv != null) {
        bal.slv({ chcBit: { idx: 'update-choice' } })
    }
    return cpy
}

export const openChoice = (cpy: ChoiceModel, bal: ChoiceBit, ste: State) => {
    const terminal = ste?.value?.terminal
    const blessed = terminal?.blessed
    const screen = terminal?.screen

    // Defensive headless fallback for automated runners and CI environments
    if (!screen || !blessed) {
        if (bal?.slv != null) {
            bal.slv({
                chcBit: {
                    idx: 'open-choice-headless',
                    src: bal?.lst && bal.lst.length > 0 ? bal.lst[0] : '',
                    val: 0,
                },
            })
        }
        return cpy
    }

    const dat: any = { idx: 'choice-bit', clr0: Color.BLACK, clr1: Color.YELLOW }
    for (const key in bal?.dat) {
        dat[key] = bal.dat[key]
    }

    const net: NetBit = bal?.net || {
        left: 0,
        top: 0,
        width: '100%',
        height: '100%',
    }

    const form = blessed.form({
        parent: screen,
        keys: true,
        mouse: true,
        left: net.left,
        top: net.top,
        width: net.width,
        height: net.height,
        bg: dat.clr0,
        content: '',
    })

    const buttons: any[] = []
    const itemList: string[] = bal?.lst || []

    itemList.forEach((itemText: string, index: number) => {
        const btn = blessed.button({
            parent: form,
            mouse: true,
            keys: true,
            shrink: true,
            padding: {
                left: 2,
                right: 1,
            },
            left: 0,
            top: index,
            height: 1,
            width: '100%',
            name: itemText,
            content: itemText,
            style: {
                bg: dat.clr1,
                focus: {
                    bg: 'red',
                },
                hover: {
                    bg: 'red',
                },
            },
        })

        btn.on('press', () => {
            form.submit()
        })

        btn.on('focus', () => {
            if (dat.cb) {
                dat.cb(itemText)
            }
        })

        buttons.push(btn)
    })

    // Vertical keyboard navigation mappings (Up/Down + Vim k/j)
    const handleUp = () => form.focusPrevious()
    const handleDown = () => form.focusNext()

    screen.key(['up', 'k'], handleUp)
    screen.key(['down', 'j'], handleDown)

    let cleanedUp = false
    const cleanup = () => {
        if (cleanedUp) return
        cleanedUp = true

        // 1. Explicitly unbind all registered directional navigation keys
        screen.unkey(['up', 'k'], handleUp)
        screen.unkey(['down', 'j'], handleDown)

        // 2. Detach and destroy widget subtree to liberate terminal memory
        form.destroy()

        // 3. Request coordinate repaint
        screen.render()
    }

    form.on('submit', () => {
        let selected = form._selected
        if (!selected && buttons.length > 0) {
            selected = buttons[0]
        }

        const src = selected ? selected.content : ''
        const val = selected ? selected.index - 1 : 0

        cleanup()

        if (bal?.slv != null) {
            bal.slv({ chcBit: { idx: 'open-choice', src, val } })
        }
    })

    form.on('cancel', () => {
        cleanup()

        if (bal?.slv != null) {
            bal.slv({ chcBit: { idx: 'open-choice-cancel', src: '', val: -1 } })
        }
    })

    if (buttons.length > 0 && typeof buttons[0].focus === 'function') {
        buttons[0].focus()
    }

    screen.render()
    return cpy
}

export const keyChoice = (cpy: ChoiceModel, bal: ChoiceBit, ste: State) => {
    const terminal = ste?.value?.terminal
    const blessed = terminal?.blessed
    const screen = terminal?.screen

    if (!screen || !blessed) {
        if (bal?.slv != null) {
            bal.slv({ scnBit: { idx: 'key-choice-headless' } })
        }
        return cpy
    }

    const net: NetBit = bal?.net || { left: 0, top: 0, width: '100%', height: '100%' }

    blessed.listbar({
        parent: screen,
        keys: true,
        left: net.left,
        top: net.top,
        width: net.width,
        height: net.height,
        style: { item: { fg: 'yellow' }, selected: { fg: 'yellow' } },
        commands: {
            Exit: {
                keys: ['C-c', 'escape'],
                callback: () => process.exit(0),
            },
        },
    })

    screen.render()
    if (bal?.slv != null) {
        bal.slv({ scnBit: { idx: 'key-choice' } })
    }
    return cpy
}

export const towerChoice = (cpy: ChoiceModel, bal: ChoiceBit, _ste: State) => {
    if (bal?.slv != null) {
        bal.slv({ chcBit: { idx: 'tower-choice' } })
    }
    return cpy
}

test('openChoice -- dynamic leak test: verifies 0 listener accumulation across 100 cycles', (t) => {
    const listeners: Record<string, Function[]> = {}

    const keyFake = (keys: string[], fn: Function) => {
        keys.forEach((k) => {
            listeners[k] = listeners[k] || []
            listeners[k].push(fn)
        })
    }

    const unkeyFake = (keys: string[], fn: Function) => {
        keys.forEach((k) => {
            if (listeners[k]) {
                listeners[k] = listeners[k].filter((h) => h !== fn)
            }
        })
    }

    let destroyedCount = 0

    const mockScreen = {
        key: keyFake,
        unkey: unkeyFake,
        render: sinon.fake(),
    }

    const makeSte = (onSubmitTrigger: { current: Function | null }) =>
        ({
            value: {
                terminal: {
                    blessed: {
                        form: () => ({
                            _selected: { content: 'TEST', index: 1 },
                            destroy: () => {
                                destroyedCount++
                            },
                            on: (event: string, cb: Function) => {
                                if (event === 'submit') onSubmitTrigger.current = cb
                            },
                        }),
                        button: () => ({
                            on: sinon.fake(),
                            focus: sinon.fake(),
                        }),
                    },
                    screen: mockScreen,
                },
            },
        }) as any

    const totalCycles = 100

    for (let i = 0; i < totalCycles; i++) {
        const onSubmitTrigger = { current: null as Function | null }
        const ste = makeSte(onSubmitTrigger)

        openChoice(
            new ChoiceModel(),
            {
                idx: `cycle-${i}`,
                lst: ['CHOICE 1', 'CHOICE 2'],
                slv: sinon.fake(),

            ste,

        if (onSubmitTrigger.current) {
            onSubmitTrigger.current()
        }
    }

    const upListeners = listeners['up']?.length || 0
    const downListeners = listeners['down']?.length || 0
    const kListeners = listeners['k']?.length || 0
    const jListeners = listeners['j']?.length || 0

    t.is(upListeners, 0, 'No lingering "up" listeners permitted after 100 iterations')

    t.is(kListeners, 0, 'No lingering "k" listeners permitted after 100 iterations')
    t.is(jListeners, 0, 'No lingering "j" listeners permitted after 100 iterations')
    t.is(destroyedCount, totalCycles, 'Every allocated form widget must be destroyed')
