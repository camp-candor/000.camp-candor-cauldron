### II. ARCHITECTURAL MISSION & REQUIREMENTS

In Blessed curses applications, when switching between complex choice menus, multi-page lists with `MORE` paginators, and input boxes (`OPEN_INPUT`), partial remnants of previous widgets remain in the terminal cell buffer if the screen is not explicitly cleared prior to mounting the next view[cite: 1]. Overlapping widgets cause cursor flickering, misaligned text borders, and phantom focus captures[cite: 1, 43].

Your mission is to harden the menu navigation controllers against visual tearing:
1. **Defensive Screen Clearing in `menu.library.ts`:**
   - Ensure every branch in `libraryMenu` that transitions to an input prompt (`LORE_GEARS`, `CREATE_GEARS`, `CREATE_UNIT`, `UPDATE_UNIT`) executes `await ste.hunt(ActTrm.CLEAR_TERMINAL, {})` before configuring the grid and mounting `OPEN_INPUT`[cite: 1, 9].
   - Ensure branches rendering nested choice forms (`UPDATE_ACTION`, `FLATTEN_UNIT`, `PROGRESS_LIBRARY`) clear the screen before mounting intermediate choice forms[cite: 1].
   - Clear the screen when transitioning back to `'ROOT MENU'` and when closing the terminal in `default`[cite: 1, 9].
2. **Defensive Sub-Menu Switching in `menu.buzz.ts`:**
   - In `updateMenu`, clear the screen (`ActTrm.CLEAR_TERMINAL`) prior to dispatching `ActMnu.LIBRARY_MENU`, dynamic pivot routes, or `ActTrm.CLOSE_TERMINAL`[cite: 1, 9].
3. **Automated Teardown Regression Battery (`fleet.menu.test.ts`):**
   - Add unit tests verifying that `ActTrm.CLEAR_TERMINAL` is reliably dispatched during sub-dialog transitions, return to root menu, and route switching[cite: 7].

            bit = await ste.hunt(ActMnu.LIBRARY_MENU, {})

                await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
                bit = await ste.hunt(ActTrm.CLOSE_TERMINAL, {})
            }
            break
        }
    }

    setTimeout(() => {
        updateMenu(cpy, bal, ste)
    }, 333)

    return cpy
}

                ySpan: 2,

                src,

        case ActUnt.FLATTEN_UNIT.split(']')[1]: {

            const isRepoRoot = (dir: string) => {
                try {
                    return (
                        FS.existsSync(path.join(dir, 'apps')) &&
                        FS.existsSync(path.join(dir, 'packages')) &&
                        FS.existsSync(path.join(dir, 'package.json'))
                    )
                } catch {
                    return false
                }
            }

            let repoRoot = process.cwd()
            while (repoRoot && !isRepoRoot(repoRoot)) {
                const parent = path.dirname(repoRoot)
                if (parent === repoRoot) break
                repoRoot = parent
            }

            lst = ['apps', 'packages']
            bit = await ste.hunt(ActGrd.UPDATE_GRID, {
                x: 0,
                y: 4,
                xSpan: 4,
                ySpan: 12,

                dat: { clr0: Color.BLACK, clr1: Color.YELLOW },
                src: Align.VERTICAL,

                net: bit.grdBit.dat,

            src = bit.chcBit.src
            if (!src) break
            const targetFolder = src.replace(/[[\]]/g, '').trim()
            const targetPath = path.join(repoRoot, targetFolder)

            const IGNORE_MENU_DIRS = new Set([
                'node_modules',
                'dist',
                'data',
                '.git',
            ])
            let childDirs: string[] = []

            try {
                if (FS.existsSync(targetPath)) {
                    const entries = FS.readdirSync(targetPath, {
                        withFileTypes: true,
                    })
                    for (const entry of entries) {
                        if (
                            entry.isDirectory() &&
                            !IGNORE_MENU_DIRS.has(entry.name)
                        ) {
                            childDirs.push(entry.name)
                        }
                    }
                }
            } catch {
                // ignore read error
            }

            childDirs.sort((a, b) => a.localeCompare(b))

            if (childDirs.length === 0) {
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `No directories found in ${targetFolder}`,
                })
                break

            dex = 0

                bit = await ste.hunt(ActGrd.UPDATE_GRID, {
                    x: 0,
                    y: 4,
                    xSpan: 4,
                    ySpan: 12,

                    dat: { clr0: Color.BLACK, clr1: Color.YELLOW },
                    src: Align.VERTICAL,
                    lst: list,
                    net: bit.grdBit.dat,

                src = bit.chcBit.src
                if (src === 'MORE') {
                    dex = dex + 10 >= childDirs.length ? 0 : dex + 10
                    await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
                } else {
                    childSelection = src
                }

            if (!childSelection) break
            const selectedDir = childSelection.replace(/[[\]]/g, '').trim()
            const selectedFullPath = path.join(targetPath, selectedDir)

            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `Flattening ${targetFolder}/${selectedDir}... Please wait.`,
            })

            const flattenUnitBit = await ste.hunt(ActUnt.FLATTEN_UNIT, {
                idx: selectedDir,
                src: selectedFullPath,
            })

            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `Flattened to ${flattenUnitBit?.untBit?.src || selectedDir}`,

            bit = await ste.hunt(ActMnu.PRINT_MENU, flattenUnitBit)
            break

                idx: 'cns00',
                src: 'Flattening library... Please wait.',

                idx: 'cns00',
                src:

            })
            bit = await ste.hunt(ActMnu.PRINT_MENU, bit)
            break

            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActLib.LIST_LIBRARY, {})
            lst = bit.libBit.lst

            if (lst.length === 0) {
                bit = await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: 'No Libraries Found',
                })

                    idx: 'cns00',
                    src: 'Listing Libraries...',
                })
                lst.forEach((a: string) =>
                    ste.hunt(ActCns.UPDATE_CONSOLE, { idx: 'cns00', src: a }),
                )
            }

            await new Promise((resolve) => setTimeout(resolve, 3000))
            break
        }

        case ActLib.AUDIT_LIBRARY.split(']')[1]: {
            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: '>> Executing Fleet Drift Audit...',

            bit = await ste.hunt(ActLib.AUDIT_LIBRARY, {})

            if (data) {
                const { totalTracked, alignedCount, driftedCount } = data
                ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `--- AUDIT SUMMARY ---`,
                })
                ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `Total Repositories: ${totalTracked}`,
                })
                ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `Aligned: ${alignedCount}`,
                })
                ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `Drifted: ${driftedCount}`,
                })
            }

            await new Promise((resolve) => setTimeout(resolve, 333))
            bit = await ste.hunt(ActMnu.PRINT_MENU, bit)
            break
        }

        case ActLib.UPDATE_LIBRARY.split(']')[1]: {
            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActLib.LIST_LIBRARY, {})
            const fullLibList = bit.libBit.lst
            const displayLibList = fullLibList.map((item: string) => {
                const clean = item.replace(/[[\]]/g, '')
                const parts = clean.split('/')
                return `[${parts[parts.length - 1]}]`
            })

            let libSelection = ''
            dex = 0
            while (libSelection === '') {
                const items = displayLibList.slice(dex, dex + 10)
                const list = [...items]
                if (displayLibList.length > 10) list.push('MORE')

                bit = await ste.hunt(ActGrd.UPDATE_GRID, {
                    x: 0,
                    y: 4,
                    xSpan: 4,
                    ySpan: 12,
                })
                bit = await ste.hunt(ActChc.OPEN_CHOICE, {
                    dat: { clr0: Color.BLACK, clr1: Color.YELLOW },
                    src: Align.VERTICAL,
                    lst: list,
                    net: bit.grdBit.dat,
                })

                src = bit.chcBit.src
                if (src === 'MORE') {
                    dex = dex + 10 >= displayLibList.length ? 0 : dex + 10
                    await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
                } else {
                    const selectedIdx = displayLibList.indexOf(src)
                    libSelection = fullLibList[selectedIdx]
                }
            }

            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActLib.UPDATE_LIBRARY, {
                src: libSelection,
            })
            bit = await ste.hunt(ActMnu.PRINT_MENU, bit)
            break
        }

        case ActLib.PROGRESS_LIBRARY.split(']')[1]: {
            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActLib.SCAN_LIBRARY, {})
            lst = bit.libBit.lst

            bit = await ste.hunt(ActGrd.UPDATE_GRID, {
                x: 0,
                y: 4,
                xSpan: 4,
                ySpan: 12,
            })
            bit = await ste.hunt(ActChc.OPEN_CHOICE, {
                dat: { clr0: Color.BLACK, clr1: Color.YELLOW },
                src: Align.VERTICAL,
                lst,
                net: bit.grdBit.dat,
            })
            src = bit.chcBit.src

            if (src) {
                await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: 'Progressing library... Please wait.',
                })
                bit = await ste.hunt(ActLib.PROGRESS_LIBRARY, { src })
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: 'Library progression completed.',
                })
            } else {
                ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: 'No directory selected',
                })
            }

            bit = await ste.hunt(ActMnu.PRINT_MENU, bit)
            break
        }

        case ActMnu.UPDATE_MENU.split(']')[1]: {
            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActMnu.UPDATE_MENU, {})
            bit = await ste.hunt(ActMnu.PRINT_MENU, bit)
            break
        }

        case 'ROOT MENU': {
            // Defensive teardown prior to returning to master navigation switchboard
            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActMnu.UPDATE_MENU, {})
            return cpy
        }

        default: {
            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActTrm.CLOSE_TERMINAL, {})
            break
        }
    }

    setTimeout(async () => {
        bit = await ste.hunt(ActMnu.LIBRARY_MENU, {})
    }, 333)

    return cpy
}
