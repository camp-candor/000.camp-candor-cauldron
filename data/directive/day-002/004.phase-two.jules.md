```typescript
import DoctorBit, { DiagnosticReportItem } from '../fce/doctor.bit.js';
import { DoctorModel } from '../doctor.model.js';
import State from '../../99.core/state.js';

export const probeRuntime = (cpy: DoctorModel, bal: DoctorBit, ste: State): DoctorModel => {
  const version = process.version;
  const major = parseInt(version.replace('v', '').split('.')[0], 10);
  const passed = major >= 20;

  const result: DiagnosticReportItem = {
    id: 'node_runtime',
    name: 'Node.js Engine',
    status: passed ? 'PASS' : 'FAIL',
    details: `${version} (Requires Node.js >= v20.0.0)`
  };

  if (bal.slv) bal.slv({ docBit: { idx: 'probe-runtime', val: passed ? 1 : 0, dat: result } });
  return cpy;
};
```

```typescript
import { execSync } from 'child_process';
import DoctorBit, { DiagnosticReportItem } from '../fce/doctor.bit.js';
import { DoctorModel } from '../doctor.model.js';
import State from '../../99.core/state.js';

export const probeQuantizer = (cpy: DoctorModel, bal: DoctorBit, ste: State): DoctorModel => {
  let passed = false;
  let details = 'ffprobe binary missing from host PATH';

  try {
    const stdout = execSync('ffprobe -version', {
      encoding: 'utf8',
      stdio: ['pipe', 'pipe', 'ignore'],
      timeout: 3000
    });
    details = stdout.split('\n')[0].replace(/[^\x20-\x7E]/g, '').trim();
    passed = true;
  } catch {
    passed = false;
  }

  const result: DiagnosticReportItem = {
    id: 'acoustic_quantizer',
    name: 'ffprobe Frame Quantizer',
    status: passed ? 'PASS' : 'WARN',
    details
  };

  if (bal.slv) bal.slv({ docBit: { idx: 'probe-quantizer', val: passed ? 1 : 0, dat: result } });
  return cpy;
};

```

```typescript
import * as os from 'os';
import DoctorBit, { DiagnosticReportItem } from '../fce/doctor.bit.js';
import { DoctorModel } from '../doctor.model.js';
import State from '../../99.core/state.js';

export const probeMemory = (cpy: DoctorModel, bal: DoctorBit, ste: State): DoctorModel => {
  const totalBytes = os.totalmem();
  const totalGb = Math.round((totalBytes / (1024 * 1024 * 1024)) * 10) / 10;
  const compliant = totalGb >= 8.0;

  const result: DiagnosticReportItem = {
    id: 'viewport_ram',
    name: 'Headless Viewport Memory',
    status: compliant ? 'PASS' : 'WARN',
    details: `${totalGb} GB Total Host RAM (Recommended >= 8.0 GB)`
  };

  if (bal.slv) bal.slv({ docBit: { idx: 'probe-memory', val: compliant ? 1 : 0, dat: result } });
  return cpy;
};
```

```typescript
import DoctorBit, { DiagnosticReportItem } from '../fce/doctor.bit.js';
import { DoctorModel } from '../doctor.model.js';
import State from '../../99.core/state.js';
import { probeRuntime } from './probeRuntime.buzz.js';
import { probeQuantizer } from './probeQuantizer.buzz.js';
import { probeMemory } from './probeMemory.buzz.js';

export const runDoctor = async (cpy: DoctorModel, bal: DoctorBit, ste: State): Promise<DoctorModel> => {
  const reports: DiagnosticReportItem[] = [];

  const logConsole = async (msg: string) => {
    if (global.LIBRARY && typeof global.LIBRARY.hunt === 'function') {
      try {
        await global.LIBRARY.hunt('[Console action] Update Console', {
          idx: 'cns00',
          src: msg
        });
      } catch {
        // Fallback for non-interactive runner
      }
    }
  };

  await logConsole('>> [CAULDRON DOCTOR] Initiating pre-flight hardware probes...');

  // 1. Probe Node.js Engine
  let r1: DiagnosticReportItem | null = null;
  probeRuntime(cpy, {
    idx: 'p1',
    slv: (res: any) => { r1 = res?.docBit?.dat; }
  }, ste);
  if (r1) {
    reports.push(r1);
    await logConsole(`>> [${(r1 as DiagnosticReportItem).status}] ${(r1 as DiagnosticReportItem).name}: ${(r1 as DiagnosticReportItem).details}`);
  }

  // 2. Probe ffprobe Audio Quantizer
  let r2: DiagnosticReportItem | null = null;
  probeQuantizer(cpy, {
    idx: 'p2',
    slv: (res: any) => { r2 = res?.docBit?.dat; }
  }, ste);
  if (r2) {
    reports.push(r2);
    await logConsole(`>> [${(r2 as DiagnosticReportItem).status}] ${(r2 as DiagnosticReportItem).name}: ${(r2 as DiagnosticReportItem).details}`);
  }

  // 3. Probe Host Memory
  let r3: DiagnosticReportItem | null = null;
  probeMemory(cpy, {
    idx: 'p3',
    slv: (res: any) => { r3 = res?.docBit?.dat; }
  }, ste);
  if (r3) {
    reports.push(r3);
    await logConsole(`>> [${(r3 as DiagnosticReportItem).status}] ${(r3 as DiagnosticReportItem).name}: ${(r3 as DiagnosticReportItem).details}`);
  }

  const allPassed = reports.every((r) => r.status !== 'FAIL');
  cpy.lastRunPassed = allPassed;
  cpy.reports = reports;
  cpy.evaluatedAt = new Date().toISOString();

  await logConsole(`>> [CAULDRON DOCTOR] Pre-flight assessment complete: [${allPassed ? 'PASS' : 'FAIL'}]`);

  if (bal.slv) {
    bal.slv({
      docBit: {
        idx: 'run-doctor',
        val: allPassed ? 1 : 0,
        dat: reports
      }
    });
  }
  return cpy;
};
```
