```typescript
// scratch/test-scaffold-adversarial.ts
import path from 'path';
import fs from 'fs-extra';
import { UnitModel } from '../apps/995.library/995.library/01.unit.unit/unit.model';
import { createUnit, updateUnit } from '../apps/995.library/995.library/01.unit.unit/buz/unit.buzz';

async function runAdversarialBattery() {
    console.log('>> [AUDIT] Running Phase 2 Scaffolding Negative-Control Battery...');
    const repoRoot = process.cwd();
    const scratchDir = path.join(repoRoot, 'scratch_adversarial_test');
    await fs.ensureDir(scratchDir);

    try {
        // --- TEST 1: Zero-Delay Execution Speed ---
        console.log('>> [AUDIT] Test 1: Asserting immediate promise resolution (<500ms)...');
        let slvCalled = false;
        const bal1 = {
            idx: '99.telemetry',
            src: scratchDir,
            slv: (res: any) => {
                slvCalled = true;
            }
        };

        const t0 = Date.now();
        await createUnit(new UnitModel(), bal1 as any, null as any);
        const elapsed = Date.now() - t0;

        if (!slvCalled) throw new Error('bal.slv was not executed by createUnit');
        if (elapsed >= 1000) throw new Error(`Execution exceeded delay limit: took ${elapsed}ms (legacy sleep detected)`);
        console.log(`>> [OK] Test 1 passed: scaffolded in ${elapsed}ms`);

        // --- TEST 2: Direct Target Path Verification ---
        console.log('>> [AUDIT] Test 2: Verifying direct filesystem placement...');
        const expectedDir = path.join(scratchDir, '99.telemetry.unit');
        if (!fs.existsSync(expectedDir)) {
            throw new Error(`Target unit was not emitted into designated src: ${expectedDir}`);
        }

        const requiredFiles = [
            'telemetry.action.ts',
            'telemetry.buzzer.ts',
            'telemetry.model.ts',
            'telemetry.reduce.ts',
            'telemetry.unit.ts',
            'buz/telemetry.buzz.ts',
            'fce/telemetry.interface.ts',
            'fce/telemetry.bit.ts'
        ];

        for (const file of requiredFiles) {
            const p = path.join(expectedDir, file);
            if (!fs.existsSync(p)) throw new Error(`Missing scaffolded file: ${file}`);
            const content = await fs.readFile(p, 'utf8');
            if (content.includes('{{=it.')) throw new Error(`Unrendered template variable found in ${file}`);
        }
        console.log('>> [OK] Test 2 passed: all 8 unit artifacts rendered cleanly in destination');

        // --- TEST 3: Negative Control - Malformed/Non-Existent Target Path Handling ---
        console.log('>> [AUDIT] Test 3: Negative control - updateUnit against non-existent unit...');
        let errReported = false;
        const balErr = {
            idx: 'nonexistent.unit',
            src: scratchDir,
            slv: (res: any) => {
                if (res?.untBit?.idx === 'update-unit-error') {
                    errReported = true;
                }
            }
        };

        await updateUnit(new UnitModel(), balErr as any, null as any);
        if (!errReported) throw new Error('updateUnit failed to return safe error bit on missing target');
        console.log('>> [OK] Test 3 passed: gracefully handled invalid unit update');

    } finally {
        await fs.remove(scratchDir).catch(() => {});
    }

    console.log('>> [OK] All Phase 2 adversarial checks certified.');
}

runAdversarialBattery().catch((err) => {
    console.error(':: [FAIL] Adversarial battery failed:', err);
    process.exit(1);
});
```
