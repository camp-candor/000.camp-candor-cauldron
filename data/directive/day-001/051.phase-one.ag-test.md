```typescript
import { ChoiceModel } from '../apps/995.library/995.library/85.choice.unit/choice.model';
import { openChoice } from '../apps/995.library/995.library/85.choice.unit/buz/choice.buzz';

async function testReentrantListenerLeak() {
    console.log('>> [AUDIT] Testing Choice Unit Listener Lifecycle across 100 re-entrant ticks...');

    const listeners: Record<string, Function[]> = {};
    const keyFake = (keys: string[], fn: Function) => {
        keys.forEach(k => {
            listeners[k] = listeners[k] || [];
            listeners[k].push(fn);
        });
    };
    const unkeyFake = (keys: string[], fn: Function) => {
        keys.forEach(k => {
            if (listeners[k]) {
                listeners[k] = listeners[k].filter(h => h !== fn);
            }
        });
    };

    let destroyedWidgets = 0;
    const mockScreen = {
        key: keyFake,
        unkey: unkeyFake,
        render: () => {},
    };

    const makeSte = () => ({
        value: {
            terminal: {
                blessed: {
                    form: () => ({
                        _selected: { content: 'TEST', index: 1 },
                        destroy: () => { destroyedWidgets++; },
                        on: (event: string, cb: Function) => {
                            if (event === 'submit') setTimeout(cb, 2);
                        },
                    }),
                    button: () => ({ on: () => {} }),
                },
                screen: mockScreen,
            },
        },
    } as any);

    for (let i = 0; i < 100; i++) {

        });

    const downListeners = listeners['down']?.length || 0;

    console.log('>> [OK] Zero listener accumulation confirmed across 100 cycles.');
```

Confirm that high-velocity telemetry bursts do not monopolize libuv or starve timer/I/O phases.

```typescript
// scratch/test-console-throttle.ts
import { ConsoleModel } from '../apps/995.library/995.library/83.console.unit/console.model';
import { updateConsole } from '../apps/995.library/995.library/83.console.unit/buz/console.buzz';

async function testConsoleEventLoopYield() {
    console.log('>> [AUDIT] Measuring event-loop latency during 500-line telemetry burst...');

    let renderCount = 0;
    let logBufferLines = 0;

    const mockTerminal = {
        screen: {
            render: () => { renderCount++; },
            cols: 80,
        },
    };

    const mockLogWidget = {
        width: 60,
        log: () => { logBufferLines++; },
    };

    const ste = {
        value: { terminal: mockTerminal },
        hunt: async (action: string) => {
            if (action.includes('Read Console')) {
                return { cnsBit: { dat: { bit: mockLogWidget } } };
            }
            return {};
        },
    } as any;

    const startTime = Date.now();
    let maxLagMs = 0;

    // Background timer to monitor event loop latency
    const interval = setInterval(() => {
        const now = Date.now();
        const lag = now - lastTick - 10;
        if (lag > maxLagMs) maxLagMs = lag;
        lastTick = now;
    }, 10);
    let lastTick = Date.now();

    // Push 500 log lines in a rapid burst
    const burstPromises = [];
    for (let i = 0; i < 500; i++) {
        burstPromises.push(
            updateConsole(new ConsoleModel(), {
                idx: 'cns00',
                src: `Telemetry receipt frame #${i}: status=[OK] verify=pass`,
            } as any, ste)
        );
    }

    await Promise.all(burstPromises);
    
    // Allow scheduled micro-tasks to settle
    await new Promise((resolve) => setTimeout(resolve, 50));
    clearInterval(interval);

    console.log(`>> [AUDIT] 500 lines processed into ${logBufferLines} widget entries.`);
    console.log(`>> [AUDIT] Actual screen repaints executed: ${renderCount}`);
    console.log(`>> [AUDIT] Max event loop lag observed: ${maxLagMs}ms`);

    // Invariants:
    // 1. Render count must be coalesced (significantly less than 500, typically < 10)
    // 2. Event loop lag should remain below 50ms
    if (renderCount >= 500) {
        console.error(':: [FAIL] Render calls were not decoupled! Synchronous diffing detected.');
        process.exit(1);
    }

    if (maxLagMs > 50) {
        console.warn(`:: [WARN] Elevated event loop lag: ${maxLagMs}ms`);
    }

    console.log('>> [OK] Event-loop decoupling and batch render coalescing verified.');
}

testConsoleEventLoopYield().catch((err) => {
    console.error(':: [FATAL]', err);
    process.exit(1);
});
```
