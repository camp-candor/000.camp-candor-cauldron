const fs = require('fs');
const path = require('path');

const emojiRegex = /[\u{1F300}-\u{1FAD6}\u{1F900}-\u{1F9FF}\u{2600}-\u{26FF}\u{2700}-\u{27BF}]/u;
const scanDirs = ['apps/995.library', 'packages', 'AGENTS.md'];
let violations = [];

function scan(target) {
    if (!fs.existsSync(target)) return;
    const stat = fs.statSync(target);
    if (stat.isDirectory()) {
        if (target.includes('node_modules') || target.includes('dist') || target.includes('.git')) return;
        for (const entry of fs.readdirSync(target)) {
            scan(path.join(target, entry));
        }
    } else if (stat.isFile()) {
        const content = fs.readFileSync(target, 'utf8');
        const lines = content.split(/\r?\n/);
        lines.forEach((line, idx) => {
            const match = line.match(emojiRegex);
            if (match) {
                violations.push({
                    file: target,
                    line: idx + 1,
                    char: match[0],
                    hex: '\\u{' + match[0].codePointAt(0).toString(16) + '}',
                    snippet: line.trim()
                });
            }
        });
    }
}

scanDirs.forEach(scan);

if (violations.length > 0) {
    console.error(':: [FAIL] UTF-8 MULTI-BYTE EMOJIS FOUND (' + violations.length + ' instances):');
    violations.forEach(v => {
        console.error('   -> ' + v.file + ':' + v.line + ' [' + v.hex + '] \"' + v.snippet + '\"');
