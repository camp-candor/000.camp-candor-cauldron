
# TASK SPECIFICATION: Add [ALL SIBLINGS] Batch Propagation to Progress Library in 995.library

## 1. Context & Objectives
In `apps/995.library/995.library/98.menu.unit/buz/menu.library.ts`, selecting `Progress Library` (`ActLib.PROGRESS_LIBRARY`) discovers sibling repositories hosting `apps/995.library` via `ActLib.SCAN_LIBRARY`. Currently, it only allows single-target sequential selection.

This task updates the `Progress Library` flow to:
1. Prepend `[ALL SIBLINGS]` to the discovered target list.
2. Provide a two-phase confirmation gate (`[NO] CANCEL BATCH SYNC` vs `[YES] BROADCAST TO N SIBLINGS`) with default focus clamped to `[NO]`.
3. Route batch synchronization through the transactional `ActLib.PROGRESS_LIBRARY_SAGA` engine (`progressLibrarySaga`), ensuring pre-flight git tree checks, zero-trust file filtration, and atomic inode swaps are applied to all targets.
4. Route any encountered torn-state or file-lock errors to `ActMnu.SAGA_TORN_STATE_MENU`.
5. Maintain strict 7-bit pure ASCII formatting for all console outputs to `cns00`.

---

## 2. Invariants & Guardrails
- **7-Bit Pure ASCII Compliance:** Strictly avoid multi-byte UTF-8 emojis. Use tokens like `>>`, `[OK]`, `[FAIL]`, `[SAGA]`, `::`.
- **Viewport Logging Isolation:** Never emit `console.log` or `console.error` while Blessed curses is active. Route all execution telemetry to `cns00` via `ste.hunt(ActCns.UPDATE_CONSOLE, { idx: 'cns00', src })`.
- **Atomic Screen Teardown:** Call `ste.hunt(ActTrm.CLEAR_TERMINAL, {})` before mounting and after dismissing choice widgets.
- **Fail-Closed Safety:** If `PROGRESS_LIBRARY_SAGA` returns `progress-library-saga-error`, escalate directly to `ActMnu.SAGA_TORN_STATE_MENU` with `dat: sagaRes.libBit.dat`.

### Step 1: Update `apps/995.library/995.library/98.menu.unit/buz/menu.library.ts`

```typescript
        case ActLib.PROGRESS_LIBRARY.split(']')[1]: {
            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})
            bit = await ste.hunt(ActLib.SCAN_LIBRARY, {})
            const targets: string[] = bit?.libBit?.lst || []

            if (targets.length === 0) {
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: '>> [FLEET NOTICE] No sibling repositories discovered.',
                })
                bit = await ste.hunt(ActMnu.PRINT_MENU, bit)
                break
            }

            const allSiblingsLabel = `[ALL SIBLINGS (${targets.length})]`
            lst = [allSiblingsLabel, ...targets]

            const descriptions: Record<string, string> = {
                [allSiblingsLabel]: `-Atomically broadcast 995.library\nacross all ${targets.length} sibling repositories.`,
            }

            for (const t of targets) {
                descriptions[t] = `-Synchronize 995.library into\n${t}`
            }

            bit = await ste.hunt(ActGrd.UPDATE_GRID, {
                x: 0,
                y: 4,
                xSpan: 4,
                ySpan: 12,
            })
            bit = await ste.hunt(ActChc.OPEN_CHOICE, {
                dat: {
                    clr0: Color.BLACK,
                    clr1: Color.YELLOW,
                    cb: (choice: string) => {
                        const text = descriptions[choice] || 'No description available.'
                        text.split('\n').forEach((s: string) =>
                            ste.hunt(ActCns.UPDATE_CONSOLE, {
                                idx: 'cns00',
                                src: s,
                            }),
                        )
                    },
                },
                src: Align.VERTICAL,
                lst,
                net: bit.grdBit.dat,
            })
            src = bit?.chcBit?.src

            await ste.hunt(ActTrm.CLEAR_TERMINAL, {})

            if (src === allSiblingsLabel || src === '[ALL SIBLINGS]') {
                // Two-phase safety confirmation modal
                const confirmGrid = await ste.hunt(ActGrd.UPDATE_GRID, {
                    x: 0,
                    y: 4,
                    xSpan: 4,
                    ySpan: 4,
                })
                const confirmChoice = await ste.hunt(ActChc.OPEN_CHOICE, {
                    dat: { clr0: Color.BLACK, clr1: Color.RED },
                    src: Align.VERTICAL,
                    lst: [
                        '[NO]  CANCEL BATCH SYNC',
                        `[YES] BROADCAST TO ${targets.length} SIBLINGS`,
                    ],
                    net: confirmGrid.grdBit.dat,
                })

                await ste.hunt(ActTrm.CLEAR_TERMINAL, {})

                if (confirmChoice?.chcBit?.src?.startsWith('[YES]')) {
                    await ste.hunt(ActCns.UPDATE_CONSOLE, {
                        idx: 'cns00',
                        src: `>> [SAGA] Initiating batch broadcast to ${targets.length} sibling repos...`,
                    })

                    const sagaRes: any = await ste.hunt(ActLib.PROGRESS_LIBRARY_SAGA, {
                        lst: targets,
                    })

                    if (sagaRes?.libBit?.idx === 'progress-library-saga-error') {
                        await ste.hunt(ActMnu.SAGA_TORN_STATE_MENU, {
                            dat: sagaRes?.libBit?.dat,
                        })
                        return cpy
                    }

                    await ste.hunt(ActCns.UPDATE_CONSOLE, {
                        idx: 'cns00',
                        src: `>> [OK] Fleet propagation completed for ${targets.length} repositories.`,
                    })
                } else {
                    await ste.hunt(ActCns.UPDATE_CONSOLE, {
                        idx: 'cns00',
                        src: '>> [ABORT] Batch broadcast cancelled by operator.',
                    })
                }
            } else if (src) {
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `>> [SAGA] Progressing library to target: ${src}`,
                })

                const sagaRes: any = await ste.hunt(ActLib.PROGRESS_LIBRARY_SAGA, {
                    lst: [src],
                })

                if (sagaRes?.libBit?.idx === 'progress-library-saga-error') {
                    await ste.hunt(ActMnu.SAGA_TORN_STATE_MENU, {
                        dat: sagaRes?.libBit?.dat,
                    })
                    return cpy
                }

                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `>> [OK] Target synchronization complete: ${src}`,
                })
            } else {
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: '>> [NOTICE] No directory selected.',
                })
            }

            bit = await ste.hunt(ActMnu.PRINT_MENU, bit)
            break
        }

```

test.serial(
    'libraryMenu -- Progress Library prepends [ALL SIBLINGS] and dispatches PROGRESS_LIBRARY_SAGA on confirmation',
    async (t) => {
        const model = new MenuModel()
        const mockTargets = [
            '[../000.series-dashboard/apps/995.library]',
            '[../000.series-sower-engine/apps/995.library]',
        ]

        let displayedChoices: string[] = []
        let sagaDispatched = false
        let sagaPayload: any = null

        const mockState: any = {
            hunt: sinon.spy(async (actionType: string, payload: any) => {
                if (actionType === ActTrm.CLEAR_TERMINAL) return {}
                if (actionType === ActCns.UPDATE_CONSOLE) return {}
                if (actionType === ActGrd.UPDATE_GRID) {
                    return { grdBit: { dat: {} } }
                }
                if (actionType === ActChc.OPEN_CHOICE) {
                    // First choice modal: select [ALL SIBLINGS (2)]
                    if (payload.lst.some((s: string) => s.startsWith('[ALL SIBLINGS'))) {
                        displayedChoices = payload.lst
                        return { chcBit: { src: payload.lst[0] } }
                    }
                    // Second choice modal (Confirmation): select [YES]
                    if (payload.lst.some((s: string) => s.startsWith('[YES]'))) {
                        return { chcBit: { src: payload.lst[1] } }
                    }
                    // Initial library menu choice: select Progress Library
                    return { chcBit: { src: ActLib.PROGRESS_LIBRARY.split(']')[1] } }
                }
                if (actionType === ActLib.SCAN_LIBRARY) {
                    return { libBit: { lst: mockTargets } }
                }
                if (actionType === ActLib.PROGRESS_LIBRARY_SAGA) {
                    sagaDispatched = true
                    sagaPayload = payload
                    return { libBit: { idx: 'progress-library-saga', val: mockTargets.length } }
                }
                if (actionType === ActMnu.PRINT_MENU) return {}
                return {}
            }),
        }

        await libraryMenu(model, { idx: 'test-progress-batch' } as any, mockState)

        t.true(
            displayedChoices.some((c) => c.startsWith('[ALL SIBLINGS')),
            'Must prepend [ALL SIBLINGS] option to choice list',
        )
        t.true(sagaDispatched, 'Must dispatch PROGRESS_LIBRARY_SAGA when confirmed')
        t.deepEqual(
            sagaPayload?.lst,
            mockTargets,
            'Must pass all discovered targets to PROGRESS_LIBRARY_SAGA',
        )
    },
)
